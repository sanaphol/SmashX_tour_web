function login() { const userInput = document.getElementById('u').value.trim().toLowerCase(); const passInput = document.getElementById('p').value.trim(); const found = state.users.find(x => x.u.toLowerCase() === userInput && x.p === passInput ); if (found) { curUser = found; document.getElementById('loginPage').classList.add('hidden'); document.getElementById('app').classList.remove('hidden'); document.getElementById('displayUser').innerText = `@${found.u}`; renderNav(); go(curUser.r === 'admin' ? 'tabPlayers' : 'tabGroups'); } else { alert('เข้าไม่ได้! ตรวจสอบ Username/Pass อีกครั้ง'); } }
